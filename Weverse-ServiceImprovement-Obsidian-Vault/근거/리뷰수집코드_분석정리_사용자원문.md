- 구글플레이 리뷰 크롤링 코드
    - 한국어, 한국 국가 제한
    - 2021-01-01 부터 2024-11-01 까지
    - 관련성 순
    
    ```python
    !pip install google_play_scraper
    
    from google_play_scraper import Sort, reviews_all
    from datetime import datetime
    import pandas as pd
    
    # 언어 및 국가 코드 설정
    lang = 'ko'
    cont = 'kr'
    
    # 크롤링 시작
    try:
        result = reviews_all(
            'co.benx.weverse',
            sleep_milliseconds=1000,
            lang=lang,
            country=cont,
            sort=Sort.MOST_RELEVANT,
            filter_score_with=None
        )
        print(f"Total reviews fetched: {len(result)}")
    except Exception as e:
        print(f"Error fetching reviews: {e}")
        result = []
    
    # 날짜 범위 설정
    start_date_str = "2021-01-01"
    end_date_str = "2024-11-01"
    start_date = datetime.strptime(start_date_str, "%Y-%m-%d")
    end_date = datetime.strptime(end_date_str, "%Y-%m-%d")
    
    # 리뷰 데이터 필터링 및 저장
    reviews_data = []
    for item in result:
        # 데이터 유효성 검사
        if not all(k in item for k in ['at', 'userName', 'score', 'content']):
            continue
        if start_date <= item['at'] <= end_date:
            review = {
                'date': item['at'],
                'userName': item['userName'],
                'score': item['score'],
                'content': item['content']
            }
            reviews_data.append(review)
    
    # 데이터프레임 생성 및 저장
    df_reviews = pd.DataFrame(reviews_data)
    print(f"Filtered reviews count: {len(df_reviews)}")
    
    csv_filename = f"reviews_{start_date_str.replace('-', '')}_to_{end_date_str.replace('-', '')}.csv"
    df_reviews.to_csv(csv_filename, index=False, encoding='utf-8-sig')
    print(f"Reviews saved to {csv_filename}")
    
    #맥 기반 워킹 디렉토리 위치 파악
    import os
    print("Current working directory:", os.getcwd())
    
    #파일 다운받기
    from google.colab import files
    files.download('/content/reviews_20210101_to_20241101.csv')
    ```

- 앱스토어 리뷰 크롤링 코드 (긁어온게 96개 밖에 없어서 조정 필요)
    
    ```python
    import requests
    import pandas as pd
    from datetime import datetime
    
    # 앱 ID와 국가 설정
    app_id = "1456559188"  # App Store ID
    country = "kr"  # 국가 코드
    
    # 날짜 범위 설정 (Naive datetime)
    start_date = datetime(2021, 1, 1)  # 시작 날짜
    end_date = datetime(2024, 11, 1)  # 끝 날짜
    
    # URL 템플릿
    base_url = f"https://itunes.apple.com/{country}/rss/customerreviews/page={{page}}/id={app_id}/sortby=mostrecent/json"
    
    # 데이터 저장용 리스트
    reviews = []
    
    # 페이지 반복
    for page in range(1, 11):  # 최대 10페이지 크롤링
        url = base_url.format(page=page)
        response = requests.get(url)
        
        if response.status_code != 200:
            print(f"Error fetching page {page}: {response.status_code}")
            break
        
        data = response.json()
        if "feed" not in data or "entry" not in data["feed"]:
            print(f"No reviews found on page {page}")
            break
        
        for entry in data["feed"]["entry"]:
            if isinstance(entry, dict):
                reviews.append({
                    "userName": entry.get("author", {}).get("name", {}).get("label", "Unknown"),
                    "rating": int(entry.get("im:rating", {}).get("label", 0)),
                    "title": entry.get("title", {}).get("label", ""),
                    "content": entry.get("content", {}).get("label", ""),
                    "date": entry.get("updated", {}).get("label", "")  # 원본 날짜
                })
        print(f"Page {page} fetched successfully.")
    
    # 날짜 필터링 (Naive datetime 사용)
    filtered_reviews = []
    for entry in reviews:
        try:
            # 타임존 정보가 있는 datetime 객체 생성
            review_date = datetime.fromisoformat(entry["date"])
    
            # 타임존 제거 (Naive datetime으로 변환)
            review_date_naive = review_date.replace(tzinfo=None)
    
            # 날짜 범위 확인
            if start_date <= review_date_naive <= end_date:
                filtered_reviews.append(entry)
        except Exception as e:
            print(f"Error parsing date: {entry['date']} - {e}")
    
    # 데이터프레임 생성
    df_reviews = pd.DataFrame(filtered_reviews)
    
    # 저장
    csv_filename = f"appstore_reviews_{app_id}_filtered.csv"
    df_reviews.to_csv(csv_filename, index=False, encoding="utf-8-sig")
    print(f"Filtered reviews saved to {csv_filename}")
    
    #파일 다운로드
    from google.colab import files
    files.download('/content/appstore_reviews_1456559188_filtered.csv')
    ```
